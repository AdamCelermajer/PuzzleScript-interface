def idx(r,c): return r*C+c

def neighbors(i):
    r,c = divmod(i,C)
    res=[]
    if r>0: res.append(i-C)
    if r<R-1: res.append(i+C)
    if c>0: res.append(i-1)
    if c<C-1: res.append(i+1)
    return res

target1 = [i for i,v in enumerate(TARGET) if v==1]
target2 = [i for i,v in enumerate(TARGET) if v==2]

def md(i,j):
    r1,c1=divmod(i,C); r2,c2=divmod(j,C)
    return abs(r1-r2)+abs(c1-c2)

MD = [[md(i,j) for j in range(15)] for i in range(15)]

def heuristic(state):
    h=0
    for i,v in enumerate(state):
        if v==0: h += MD[i][0]
        elif v==3: h += MD[i][12]
        elif v==4: h += MD[i][13]
        elif v==5: h += MD[i][14]
        elif v==1: h += min(MD[i][t] for t in target1)
        elif v==2: h += min(MD[i][t] for t in target2)
    return h

DIRS = {-C:'U', C:'D', -1:'L', 1:'R'}
DIR_ACTION = {'U':'ACTION1','D':'ACTION2','L':'ACTION3','R':'ACTION4'}

def astar(start, max_iter=4000000):
    start=tuple(start)
    if start==TARGET: return []
    h0=heuristic(start)
    openq=[(h0,0,start)]
    g={start:0}
    parent={}
    it=0
    while openq and it<max_iter:
        it+=1
        f,gval,state=heapq.heappop(openq)
        if gval>g[state]: continue
        blank=state.index(5)

            lst=list(state)
            lst[blank],lst[nb]=lst[nb],lst[blank]
            ns=tuple(lst)
            ng=gval+1

        s=TARGET

            path.append(dirn)
            s=p
        return list(reversed(path))
    return None

1. **Cursor step:** legal iff the destination is in-bounds and `|height(dest) − height(src)| ≤ 1`. Observed: Δ0 and Δ1 steps always succeed; Δ2 drops are blocked (cursor stays, grid unchanged) and Δ2 climbs are blocked (tested explicitly, grid unchanged). The rule is therefore symmetric.
