# Game mechanics (learned)
- A1=up, A2=down, A3=left, A4=right, A7=undo, RESET=reset level
- Tiles: 1=bright floor, 2=dark floor, 0=blue marker (walkable, crate target),
  3=crate on bright, 8=crate on dark, 4=checkered goal (neutral bridge tile)
- Player display: 5=on dark, 7=on bright, 6=on goal (number = f(tile below))
- Movement: player can only step on tiles of SAME class as tile below
  (bright={0,1}, dark={2,3-crate}). Goal (4) = neutral, can enter/exit from any class.
- Crates pushable (Sokoban): player moves onto crate tile, crate moves 1 in same dir.
  Push requires player's class to match BOTH tiles it traverses (start & crate's old tile).
  Crate may cross classes freely (8<->3).
- Win: push crate onto blue marker (0). Level completes on that push.

import sys
from collections import deque

# Level 4 grid, 9 rows x 11 cols
grid = [
"22292222222",
"22212222222",
"21212122211",
"21211112211",
"73411221111",
"21211222111",
"21212222111",
"21222222111",
"21111111111",
]
R = len(grid)
C = len(grid[0])
player = crate = marker = goal = None
for r in range(R):
    for c in range(C):
        t = grid[r][c]
        if t == '7': player = (r,c)
        elif t in ('3','8'): crate = (r,c)
        elif t in ('0','9'): marker = (r,c)
        elif t == '4': goal = (r,c)

def tclass(r,c):
    t = grid[r][c]
    if t == '4': return None  # neutral
    if t in ('1','0','3'): return 'B'
    if t in ('2','9','8'): return 'D'
    return None

def inb(r,c): return 0<=r<R and 0<=c<C

DIRS = {'U':(-1,0),'D':(1,0),'L':(0,-1),'R':(0,1)}

def solve(allow_crate_on_goal=True, allow_push_from_goal=True):
    start = (player, crate)
    prev = {start: None}
    q = deque([start])
    while q:
        (pr,pc),(cr,cc) = q.popleft()
        if (cr,cc) == marker:
            # reconstruct
            path = []
            st = ((pr,pc),(cr,cc))
            while prev[st] is not None:
                st, mv = prev[st]
                path.append(mv)
            path.reverse()
            return path
        pcls = tclass(pr,pc)
        for name,(dr,dc) in DIRS.items():
            nr,nc = pr+dr, pc+dc
            if not inb(nr,nc): continue
            nt = grid[nr][nc]
            if (nr,nc) == (cr,cc):
                # push crate
                if pcls is None and not allow_push_from_goal: continue
                if pcls is not None and tclass(nr,nc) != pcls: continue
                rr,rc = nr+dr, nc+dc
                if not inb(rr,rc): continue
                rt = grid[rr][rc]
                if (rr,rc) == (pr,pc): continue
                if (rr,rc) == (cr,cc): continue
                if rt not in '1209384': continue
                if rt == '4' and not allow_crate_on_goal: continue
                nst = ((nr,nc),(rr,rc))
            else:
                if nt not in '1209384': continue
                ncls = tclass(nr,nc)
                if ncls is not None and pcls is not None and ncls != pcls: continue
                nst = ((nr,nc),(cr,cc))
            if nst not in prev:
                prev[nst] = (((pr,pc),(cr,cc)), name)
                q.append(nst)
    return None

path = solve()
print(len(path) if path else None)
print(path)
