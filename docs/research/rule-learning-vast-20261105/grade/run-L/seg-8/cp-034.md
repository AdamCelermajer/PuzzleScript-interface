def solve(grid, player0, goal, movables):
    """grid: list of rows. movables: dict value->list of positions (pushable blocks).
    STATIC_WALLS = everything not ice(6), goal(5), player(9), or in movables."""
    R,C = len(grid), len(grid[0])
    mov_vals = set(movables.keys())
    def static_open(r,c):
        if not (0<=r<R and 0<=c<C): return False
        v = grid[r][c]
        return v in (6,5,9)
    dirs={'U':(-1,0),'D':(1,0),'L':(0,-1),'R':(0,1)}
    def slide(state, d):
        (pr,pc), objs = state
        dr,dc = dirs[d]
        occ = {pos for pos in objs}
        r,c = pr,pc; moved=False
        while True:
            nr,nc = r+dr, c+dc
            if (nr,nc) in occ: break
            if not static_open(nr,nc):
                return ((r,c), objs) if moved else None
            r,c = nr,nc; moved=True
        chain=[(r+dr,c+dc)]
        while True:
            fr,fc = chain[-1]; nr,nc = fr+dr, fc+dc
            if (nr,nc) in occ: chain.append((nr,nc))
            else: break
        fr,fc = chain[-1]; dist=0
        while True:
            nr,nc = fr+dr, fc+dc
            if not static_open(nr,nc) or (nr,nc) in occ: break
            fr,fc = nr,nc; dist+=1
        if dist==0: return None
        posmap={cell:(cell[0]+dist*dr, cell[1]+dist*dc) for cell in chain}
        newobjs=tuple(sorted(posmap.get(pos,pos) for pos in objs))
        return ((fr-dr, fc-dc), newobjs)
    start=(player0, tuple(sorted(p for ps in movables.values() for p in ps)))
    seen={start}
    q=deque([(start,[])])
    while q:
        st,path=q.popleft()
        if st[0]==goal: return path, len(seen)
        for d in 'UDLR':
            ns=slide(st,d)
            if ns and ns not in seen:
                seen.add(ns); q.append((ns,path+[d]))
    return None, len(seen)
