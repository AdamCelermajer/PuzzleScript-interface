Main Runtime Loop

client/runtime/runner.py
Main loop Runtime <-> engine modules <-> ARC/PuzzleScript engine
frame ← env.reset()ARC/PuzzleScript starts the game
state ← perceiver.perceive(frame)
memory.append_state(state)
game_id ← cfg.game or state.game_id
while steps < max_steps do
decision ← planner.choose_action(state)
P ← rulebook.predict(state, decision.action)
outcome ← action_executor.execute(frame, state, decision)env.step + perceive
rulebook.record_prediction_result(state, outcome.action, outcome.after_state, P)
memory.append_action_result(outcome.action, outcome.after_state)
contradicted ← (P is not empty) and (outcome.after_state not in P)
unexplained ← (P is empty) or contradicted
if contradicted or (P is empty and not decision.exploratory) then
planner.clear_llm_plan()unknown exploratory effects can continue
if decision.exploratory or unexplained then
inducer.propose_from_memory(game_id, memory)
if state.is_win() then return success
if game over then reset env and continue
state ← outcome.after_state

Planner

client/engine/planner.py
File role Choose the next action
s ← memory.current_state()
A ← available_actions(s)exclude RESET
if active plan has next action then
active_plan.advance()
return PlanDecision(next action, active_plan, exploratory = active_plan.exploratory)
if game_goal = null then
game_goal ← goal_manager.ensure_goal(memory)
plan ← plan_to_win(s, A)BFS over rule predictions
if plan != null then
active_plan ← ActivePlan(plan, index = 1, exploratory = false)
return PlanDecision(plan[0], rule_plan)
goal ← goal_manager.ask_for_subgoal(llm, s, A, rulebook.known_rules_text())
probe ← goal_manager.ask_for_action_plan(llm, s, A, goal, rulebook.known_rules_text())short list, 1-5 actions
active_plan ← ActivePlan(probe, index = 1, exploratory = true)
return PlanDecision(probe[0], explore_subgoal, plan = probe, exploratory = true)

Rule Learning

client/engine/induction.py + client/engine/verifier.py
File role Turn fresh transitions into accepted executable rules
records ← memory.recent_transitions(1)
if records is empty then return
(events, images) ← memory.latest_visual_context()
H ← llm.call_json(natural_rule_prompt(events, rulebook.known_rules_text()), images)plain language only
if H is empty then return
response ← llm.call_json(schema_prompt(events, H), images)
candidates ← parse_rule_candidates(response)
for all c in candidates do
if verifier.verify(c).accepted then
rulebook.add_generalized_rule(c)
else log rejected candidate

Engine Wiring

client/engine/architecture.py
File role Build the reasoning modules
base_path ← game_rules_path(cfg.rules_dir, cfg.game)
memory ← EngineMemory(base_path / "timeline.jsonl")
rulebook ← Rulebook(base_path)
verifier ← RuleVerifier(memory)
inducer ← RuleInducer(llm, rulebook, verifier)
planner ← Planner(rulebook, memory, llm)
return EngineArchitecture(perceiver, memory, rulebook, inducer, planner)

Entrypoint

client/run_arc_agent.py
File role Wire one runnable session
cfg ← Config(game, max_steps, backend_url)
dashboard ← ScreenDashboard(game_id)
env ← ArcadeEnv(game_id, backend_url, dashboard.render)
llm ← LlmClient(cfg, dashboard.push_event)
engine ← EngineArchitecture.from_config(cfg, llm, dashboard.push_event)
loop ← RuleReasoningLoop(env, engine.perceiver, engine.memory, engine.rulebook, engine.planner, engine.inducer, dashboard, dashboard.push_event)
dashboard.run_engine(function: loop.run(max_steps, game_id))
